function pairSum(nums, target) {
  const seen = new Map();
  for (const [i, value] of nums.entries()) {
    const needed = target - value;
    if (seen.has(needed)) return
      [seen.get(needed), i];
    seen.set(value, i);
  }
  return null;
}
